    function handleFileUpload(e) {
      const files = e.target.files;
      if (!files || files.length === 0) return;
      processUploadedFiles(Array.from(files));

    function processUploadedFiles(filesArray) {
      let pending = filesArray.length;
      filesArray.forEach(file => {
        const cleanName = file.name.replace(/\.txt\.txt$/i, '.txt');
        const sizeFormatted = (file.size / 1024).toFixed(1) + ' KB';
        const existingIdx = activeFilesList.findIndex(f => f.name.toLowerCase() === cleanName.toLowerCase());
        if (existingIdx === -1) {
          activeFilesList.push({ name: cleanName, size: sizeFormatted });
        }
        const reader = new FileReader();
        reader.onload = function(evt) {
          uploadedDocuments[cleanName] = evt.target.result;
          pending--;
          if (pending <= 0) {
            renderActiveDocs();
            updateRubricMetrics();
            streamBotMessage(`<p class="text-white font-semibold">📁 File Indexed: <b>${cleanName}</b></p><p class="text-zinc-300 mt-1">Ready for factual queries and court document drafting.</p>`);
          }
        };
        reader.readAsText(file);
      });
      renderActiveDocs();
      updateRubricMetrics();
    }
    function removeDoc(index) {
      const removed = activeFilesList.splice(index, 1);
      if (removed && removed[0]) {
        delete uploadedDocuments[removed[0].name];
      }
      renderActiveDocs();
      updateRubricMetrics();
    }

    async function handleSend(e) {
      if (e) e.preventDefault();
      const text = userInput.value.trim();
      if (!text) return;
      appendUserMessage(text);
      userInput.value = '';
      showTypingIndicator();
      let dossierContext = '';
      for (const [name, content] of Object.entries(uploadedDocuments)) {
        if (content) dossierContext += `\n[FILE: ${name}]\n${content}\n`;
      }
      const textLower = text.toLowerCase().trim();
      const isChartIntent = textLower.includes('pie') || textLower.includes('pichart') || textLower.includes('piechart') ||
                            textLower.includes('bar chart') || textLower.includes('radar') ||
                            (textLower.includes('graph') && (textLower.includes('time') || textLower.includes('financial') || textLower.includes('all'))) ||
                            textLower.includes('data analyse') || textLower.includes('data analysis') || textLower.includes('fund flow');
      if (isChartIntent && activeFilesList.length > 0) {
        setTimeout(() => {
          removeTypingIndicator();
          const { html } = executeLocalRag(text);
          streamBotMessage(html);
        }, 300);
        return;
      }
      try {
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 25000);
        const currentProvider = localStorage.getItem('ai_provider') || 'auto';
        const currentModel = localStorage.getItem('ai_model') || (currentProvider === 'openai' ? 'gpt-4o-mini' : 'gemini-3.8-flash');
        const customKey = currentProvider === 'openai' ? (localStorage.getItem('openai_api_key') || '') : '';
        const response = await fetch(`${API_BASE}/api/chat`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            message: text,
            context: dossierContext.slice(0, 30000),
            provider: currentProvider,
            model: currentModel,
            apiKey: customKey
          }),
          signal: controller.signal
        });
        clearTimeout(timeoutId);
        if (response.ok) {
          const data = await response.json();
          if (data && (data.response || data.html)) {
            removeTypingIndicator();
            const raw = data.response || data.html;
            const formatted = raw
              .replace(/\*\*(.*?)\*\*/g, '<b>$1</b>')
              .replace(/\n\n/g, '<br><br>')
              .replace(/\n/g, '<br>');
            streamBotMessage(formatted);
            return;
          }
        }
      } catch (err) {
        console.warn('Backend server offline or loading. Using grounded conversational engine.');
      }
      setTimeout(() => {
        removeTypingIndicator();
        const { html } = executeLocalRag(text);
        streamBotMessage(html);
      }, 300);
    }
